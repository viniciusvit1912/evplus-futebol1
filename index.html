from http.server import BaseHTTPRequestHandler import json class handler(BaseHTTPRequestHandler): def do_GET(self): resposta = { "status": "online", "sistema": "EV+ Futebol", "mensagem": "API funcionando" } self.send_response(200) self.send_header("Content-Type", "application/json") self.send_header("Access-Control-Allow-Origin", "*") self.end_headers() self.wfile.write( json.dumps(resposta).encode("utf-8") ) def do_POST(self): try: tamanho = int( self.headers.get("Content-Length", 0) ) corpo = self.rfile.read(tamanho) dados = json.loads( corpo.decode("utf-8") ) probabilidade = float( dados["probabilidade"] ) odd = float( dados["odd"] ) prob = probabilidade / 100 ev = (prob * odd) - 1 valor_justo = 1 / prob resposta = { "status": "ok", "probabilidade": probabilidade, "odd": odd, "ev": round(ev * 100, 2), "valor_justo": round(valor_justo, 2) } codigo = 200 except Exception as erro: resposta = { "status": "erro", "mensagem": str(erro) } codigo = 400 self.send_response(codigo) self.send_header( "Content-Type", "application/json" ) self.send_header( "Access-Control-Allow-Origin", "*" ) self.end_headers() self.wfile.write( json.dumps(resposta).encode("utf-8") )
